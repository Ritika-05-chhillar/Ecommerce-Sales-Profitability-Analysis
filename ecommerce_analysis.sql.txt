-- 1. TOTAL REVENUE AND UNITS SOLD

CREATE OR REPLACE VIEW revenue_analysis AS
SELECT
    SUM(oi.quantity * p.unit_price * (1 - oi.discount_pct)) AS total_revenue,
    SUM(oi.quantity) AS total_units_sold
FROM order_items oi
JOIN products p
    ON oi.product_id = p.product_id
WHERE oi.quantity > 0;


-- 2. CATEGORY-WISE PROFIT

CREATE OR REPLACE VIEW category_profit AS
SELECT
    p.category,
    SUM(
        oi.quantity *
        (p.unit_price * (1 - oi.discount_pct) - p.unit_cost)
    ) AS total_profit
FROM order_items oi
JOIN products p
    ON oi.product_id = p.product_id
WHERE oi.quantity > 0
GROUP BY p.category;


-- 3. MONTHLY SALES ANALYSIS

CREATE OR REPLACE VIEW monthly_sales AS
SELECT
    TO_CHAR(DATE_TRUNC('month', o.order_date), 'YYYY-MM') AS sales_month,
    SUM(oi.quantity * p.unit_price * (1 - oi.discount_pct)) AS total_revenue
FROM orders o
JOIN order_items oi
    ON o.order_id = oi.order_id
JOIN products p
    ON oi.product_id = p.product_id
WHERE oi.quantity > 0
GROUP BY DATE_TRUNC('month', o.order_date);


-- 4. TOP PRODUCTS ANALYSIS

CREATE OR REPLACE VIEW top_products AS
SELECT
    p.product_name,
    SUM(oi.quantity) AS total_quantity_sold,
    SUM(oi.quantity * p.unit_price * (1 - oi.discount_pct)) AS total_revenue
FROM order_items oi
JOIN products p
    ON oi.product_id = p.product_id
WHERE oi.quantity > 0
GROUP BY p.product_name;


-- 5. CUSTOMER ANALYSIS

CREATE OR REPLACE VIEW customer_analysis AS
SELECT
    c.customer_id,
    c.customer_name,
    COUNT(DISTINCT o.order_id) AS total_orders,
    SUM(oi.quantity * p.unit_price * (1 - oi.discount_pct)) AS total_spent
FROM customers c
JOIN orders o
    ON c.customer_id = o.customer_id
JOIN order_items oi
    ON o.order_id = oi.order_id
JOIN products p
    ON oi.product_id = p.product_id
WHERE oi.quantity > 0
GROUP BY c.customer_id, c.customer_name;


-- 6. CATEGORY-WISE REVENUE

SELECT
    p.category,
    SUM(oi.quantity * p.unit_price * (1 - oi.discount_pct)) AS total_revenue
FROM order_items oi
JOIN products p
    ON oi.product_id = p.product_id
WHERE oi.quantity > 0
GROUP BY p.category
ORDER BY total_revenue DESC;


-- 7. DATA CLEANING CHECKS

-- Check duplicate customers
SELECT customer_id, COUNT(*)
FROM customers
GROUP BY customer_id
HAVING COUNT(*) > 1;

-- Check missing emails
SELECT *
FROM customers
WHERE email IS NULL OR TRIM(email) = '';

-- Check zero or negative quantities
SELECT *
FROM order_items
WHERE quantity <= 0;

-- Check invalid discounts
SELECT *
FROM order_items
WHERE discount_pct < 0 OR discount_pct > 1;

-- Check missing order dates
SELECT *
FROM orders
WHERE order_date IS NULL;
